\section{Background}
\label{sec:background}
\paragraph{DDPG.} \citet{lillicrap2016ddpg}: deterministic actor $\pith$, critic $\Qphi$ trained on
$y = r + \gamma (1-d)\, Q_{\phi'}(s', \pi_{\theta'}(s'))$, actor updated by $\nabla_\theta \E[\Qphi(s,\pith(s))]$.

\paragraph{TD3.} \citet{fujimoto2018td3}: (i) clipped double-Q target
$y = r + \gamma (1-d) \min_{i=1,2} Q_{\phi'_i}(s', \tilde a')$, (ii) target policy smoothing
$\tilde a' = \pi_{\theta'}(s') + \mathrm{clip}(\epsilon, -c, c)$, (iii) delayed actor updates.

\paragraph{Layer Normalization.} \citet{ba2016layernorm}: normalises pre-activations of each hidden layer over features.
\todo{Where exactly it is inserted (critic only / actor too, before or after activation).}

\paragraph{Measuring overestimation.} For a visited pair $(s_t,a_t)$, the bias is
\begin{equation}
  \Delta(s_t,a_t) = \Qphi(s_t,a_t) - G_t, \qquad G_t = \sum_{k\ge 0} \gamma^k r_{t+k},
  \label{eq:bias}
\end{equation}
where $G_t$ is the discounted Monte Carlo return obtained by running the current policy from $(s_t,a_t)$.
